funding_policy_rules [icon: settings, color: blue] {
  id string pk
  group_id string fk -> funding_policy_groups.id
  rule_type enum(
    "MAX_PERCENT",                // porcentagem máxima de contribuição
    "MIN_PERCENT",                // porcentagem mínima
    "MAX_VALUE_PER_REQUEST",      // valor máximo por solicitação
    "MIN_VALUE_PER_REQUEST",      // valor mínimo por solicitação
    "MAX_ACTIVITIES",             // quantidade máxima de atividades
    "MIN_ACTIVITIES",             // quantidade mínima
    "MAX_PROJECTS_PER_YEAR",      // máximo de projetos por ano
    "MAX_PROJECTS_PER_MONTH",     // máximo por mês
    "MIN_PROJECTS_PER_YEAR",      // mínimo de projetos anuais
    "MIN_PROJECTS_PER_MONTH"      // mínimo mensal
  )
  applies_to enum("REQUESTER","FUNDER")  // define se aplica a quem solicita ou a quem financia
  comparison_type enum("MAX","MIN")      // tipo de comparação
  value decimal                          // valor da regra (ex: 65%)
  unit enum("PERCENT","CURRENCY","COUNT") // tipo de unidade da regra
  year int nullable                      // caso a regra seja anual
  created_at timestamp
  updated_at timestamp
  created_by string
}
